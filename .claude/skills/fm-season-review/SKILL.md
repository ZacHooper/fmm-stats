# FM End of Season Review

Use this skill when the user asks for an end-of-season review, team review, or player awards for their Football Manager save parsed via `fm-parser`.

## Context
- Analytics run against the DuckDB store generated by `fm-parser`.
- The user reads these on Discord, so output must be formatted as punchy, narrative-driven text with bullet points, NOT massive markdown tables.
- The review consists of two main parts: Tactical/Positional Review (for the coaches) and Player Awards (serious & silly).

## Behavioral Guardrails (Fable Framework)

**1. The DuckDB Lock Rule**
**[REQUEST]** "Run the end of season review for Frem."
**[BAD]** *Connects directly to `fm-frem.duckdb` and crashes with an IO Error.*
**[GOOD]** *Copies `fm-frem.duckdb` to `/tmp/scout.duckdb` and connects in read-only mode.*
**[WHY]** The Streamlit dashboard often runs in the background and holds a write lock on the main career DB. Always query a `/tmp/` copy.

**2. The Starter Filtering Rule (Data Purity)**
**[REQUEST]** "Give me the positional unit breakdown for the season."
**[BAD]** *Averages ratings for all players, including 89th-minute subs.*
**[GOOD]** *Filters queries using `pos_order <= 11` and `subOn = 255`, and requires a minimum of 10 starts.*
**[WHY]** Sub appearances heavily distort average ratings and defensive/passing volume stats. Positional analysis must only evaluate pure starters.

**3. The Mobile Output Rule**
**[REQUEST]** "Who won the awards this season?"
**[BAD]** *Prints a 20-column, 15-row markdown table of Pandas DataFrame output.*
**[GOOD]** *Parses the DataFrame internally and prints formatted Discord text: "🏆 **Golden Boot:** Daniel Jørgensen (15 goals)"*
**[WHY]** Zac reads this on a phone via Discord. Wide tables break. Translate data into a tight narrative.

## Python Script Template

Use the `execute_code` tool to run this script. It connects to the `/tmp/scout.duckdb` database (ensure you copy the actual DB there first via `terminal`) and outputs the raw data needed to write the Discord response.

```python
import duckdb
import pandas as pd

# IMPORTANT: Ensure the db was copied to /tmp/scout.duckdb via terminal first
con = duckdb.connect("/tmp/scout.duckdb", read_only=True)

# Define the team TID (e.g., Frem = 346, Bucaspor = 6567). Ask user or find dynamically.
TEAM_TID = 346 

# 1. Get Latest Season
latest_season = con.execute(f"SELECT MAX(season) as s FROM staging.match_player_stats WHERE team_tid = {TEAM_TID}").df()['s'].iloc[0]
latest_phase = con.execute(f"SELECT MAX(phase) as p FROM staging.match_player_stats WHERE team_tid = {TEAM_TID} AND season = {latest_season}").df()['p'].iloc[0]

print(f"=== SEASON {latest_season} REVIEW ===")

# 2. TACTICAL REVIEW (Formations)
form_query = f\"\"\"
SELECT formation, COUNT(*) as games, 
    SUM(CASE WHEN home_tid = {TEAM_TID} THEN score_home ELSE score_away END) as scored,
    SUM(CASE WHEN home_tid = {TEAM_TID} THEN score_away ELSE score_home END) as conceded
FROM staging.matches 
WHERE (home_tid = {TEAM_TID} OR away_tid = {TEAM_TID}) AND season = {latest_season} AND phase = '{latest_phase}'
GROUP BY formation ORDER BY games DESC
\"\"\"
print("\\n--- FORMATIONS ---")
print(con.execute(form_query).df().to_string(index=False))

# 3. POSITIONAL UNIT REVIEW (Starters Only)
unit_query = f\"\"\"
WITH player_names AS (SELECT tid, ANY_VALUE(name) as name FROM staging.players GROUP BY tid),
player_pos_stats AS (
    SELECT s.tid, p.name,
        CASE 
            WHEN pos_order = 1 THEN 'GK'
            WHEN pos_order IN (2, 3) THEN 'Fullbacks'
            WHEN pos_order IN (4, 5) THEN 'Centerbacks'
            WHEN pos_order IN (6, 7, 8, 9) THEN 'Midfielders / Wingers'
            WHEN pos_order IN (10, 11) THEN 'Forwards'
        END as position_group,
        COUNT(s.anchor) as starts, ROUND(AVG(s.rating), 2) as avg_rating,
        SUM(s.goals) as goals, SUM(s.assists) as assists, SUM(s.keyPass) as key_passes,
        SUM(s.tackW + s.intercept) as def_actions, SUM(s.mistakes) as mistakes
    FROM staging.match_player_stats s
    JOIN player_names p ON s.tid = p.tid
    WHERE s.team_tid = {TEAM_TID} AND s.season = {latest_season} AND s.phase = '{latest_phase}' AND s.pos_order <= 11 AND s.subOn = 255
    GROUP BY s.tid, p.name, position_group
)
SELECT position_group, name, starts, avg_rating, goals, assists, key_passes, def_actions, mistakes
FROM player_pos_stats WHERE starts >= 10
ORDER BY CASE position_group WHEN 'GK' THEN 1 WHEN 'Centerbacks' THEN 2 WHEN 'Fullbacks' THEN 3 WHEN 'Midfielders / Wingers' THEN 4 WHEN 'Forwards' THEN 5 END, avg_rating DESC
\"\"\"
print("\\n--- PLAYERS BY UNIT (10+ Starts) ---")
print(con.execute(unit_query).df().to_string(index=False))

# 4. AWARDS
awards_query = f"""
WITH player_names AS (SELECT tid, ANY_VALUE(name) as name, ANY_VALUE({latest_season} - EXTRACT(YEAR FROM dob)) as age FROM staging.players GROUP BY tid)
SELECT p.name, p.age, COUNT(s.anchor) as games, ROUND(AVG(s.rating), 2) as avg_rating,
    SUM(s.goals) as goals, SUM(s.assists) as assists, SUM(s.headW) as headers_won,
    SUM(s.dribbles) as dribbles, SUM(s.passC) as passes_completed, SUM(s.keyPass) as key_passes,
    SUM(s.shotA) as shots_attempted, SUM(s.crossA) as crosses_attempted, SUM(s.crossC) as crosses_completed,
    SUM(s.intercept) + SUM(s.tackW) as def_actions, SUM(s.mistakes) as mistakes, SUM(s.yellow) as yellows,
    MAX(s.passC) as max_passes_in_game
FROM staging.match_player_stats s
JOIN player_names p ON s.tid = p.tid
WHERE s.team_tid = {TEAM_TID} AND s.season = {latest_season} AND s.phase = '{latest_phase}' GROUP BY p.name, p.age HAVING games > 3
"""
df = con.execute(awards_query).df()

print("\\n--- AWARDS ---")
print("Player of Season:", df.nlargest(1, 'avg_rating')[['name', 'avg_rating']].to_dict('records'))
print("Young Player (U21):", df[df['age'] <= 21].nlargest(1, 'avg_rating')[['name', 'avg_rating']].to_dict('records'))
print("Golden Boot:", df.nlargest(1, 'goals')[['name', 'goals']].to_dict('records'))
print("Top Assister:", df.nlargest(1, 'assists')[['name', 'assists']].to_dict('records'))
print("The Maestro (Key Passes):", df.nlargest(1, 'key_passes')[['name', 'key_passes']].to_dict('records'))
print("Big Head (Headers):", df.nlargest(1, 'headers_won')[['name', 'headers_won']].to_dict('records'))
print("Quick Feet (Dribbles):", df.nlargest(1, 'dribbles')[['name', 'dribbles']].to_dict('records'))
print("The Metronome (Total Passes):", df.nlargest(1, 'passes_completed')[['name', 'passes_completed']].to_dict('records'))
print("The Brick Wall (Def Actions):", df.nlargest(1, 'def_actions')[['name', 'def_actions']].to_dict('records'))
print("The Liability (Mistakes):", df.nlargest(1, 'mistakes')[['name', 'mistakes']].to_dict('records'))
print("The Butcher (Yellows):", df.nlargest(1, 'yellows')[['name', 'yellows']].to_dict('records'))

# Most Improved Player (Total Attribute Growth)
growth_query = f"""
WITH attr_totals AS (
    SELECT tid, name, phase, season,
        (COALESCE(Aerial, 0) + COALESCE(Crossing, 0) + COALESCE(Dribbling, 0) + COALESCE(Shooting, 0) + 
         COALESCE(Passing, 0) + COALESCE(Tackling, 0) + COALESCE(Technique, 0) + COALESCE(Aggression, 0) + 
         COALESCE(Creativity, 0) + COALESCE(Decisions, 0) + COALESCE(Leadership, 0) + COALESCE(Movement, 0) + 
         COALESCE(Positioning, 0) + COALESCE(Teamwork, 0) + COALESCE(Pace, 0) + COALESCE(Stamina, 0) + 
         COALESCE(Strength, 0) + COALESCE(Agility, 0) + COALESCE(Handling, 0) + COALESCE(Kicking, 0) + 
         COALESCE(Reflexes, 0) + COALESCE(Communication, 0) + COALESCE(Throwing, 0)) as total_attributes
    FROM v_player_attributes
    WHERE club_tid = {TEAM_TID} AND season = {latest_season}
)
SELECT start_data.name, (latest_data.total_attributes - start_data.total_attributes) as growth
FROM (SELECT * FROM attr_totals WHERE phase = 'start') start_data
JOIN (SELECT phase, tid, total_attributes FROM attr_totals WHERE phase != 'start' ORDER BY phase DESC) latest_data
  ON start_data.tid = latest_data.tid
ORDER BY growth DESC LIMIT 1
"""
print("Most Improved (Attribute Growth):", con.execute(growth_query).df().to_dict('records'))

# Stormtrooper (Single Match)
storm_query = f\"\"\"
WITH names AS (SELECT tid, ANY_VALUE(name) as name FROM staging.players GROUP BY tid)
SELECT n.name, s.shotA, s.date FROM staging.match_player_stats s
JOIN names n ON s.tid = n.tid
WHERE s.team_tid = {TEAM_TID} AND s.season = {latest_season} AND s.goals = 0 ORDER BY s.shotA DESC LIMIT 1
\"\"\"
print("The Stormtrooper (Match):", con.execute(storm_query).df().to_dict('records'))
```
